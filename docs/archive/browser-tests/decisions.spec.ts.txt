// Historical dashboard verification, superseded by tests/world.spec.ts and current onboarding. See docs/TEST-MIGRATION.md.
import { readFile } from 'node:fs/promises'
import { expect, test, type Page } from '@playwright/test'
import init, { TrailEngine } from '../public/wasm/pioneer_trail_web_engine.js'

// These checkpoints come from real commands against the shipped engine, never edited saves.
async function checkpoint(kind: 'river' | 'repair' | 'raft') {
  await init({ module_or_path: await readFile('public/wasm/pioneer_trail_web_engine_bg.wasm') })
  for (let seed = 11; seed < 30; seed++) {
    const game = new TrailEngine(String(seed))
    const apply = (command: unknown) => JSON.parse(game.apply(JSON.stringify(command)))
    apply({
      Configure: {
        trail_id: 'oregon',
        era_id: '1848',
        occupation_id: 'banker',
        party: ['Ada', 'James', 'Ruth', 'Thomas', 'Clara'],
        departure_month: 3,
      },
    })
    for (const [item_id, quantity] of [
      ['oxen', 3],
      ['food', 2000],
      ['clothing', 5],
      ['medicine', 3],
      ['wheel', 3],
      ['axle', 3],
      ['tongue', 3],
    ])
      apply({ Buy: { item_id, quantity } })
    apply('Depart')
    for (let step = 0; step < 400; step++) {
      const view = JSON.parse(game.view())
      if (
        (kind === 'river' && view.status.AwaitingRiver && !view.pending_event) ||
        (kind === 'repair' && view.can_repair)
      )
        return { save: game.save(), view }
      if (
        kind === 'raft' &&
        view.status.AwaitingFork &&
        view.routes.some((route: { id: string }) => route.id === 'columbia')
      ) {
        apply({ ChooseRoute: { route_id: 'columbia' } })
        return { save: game.save(), view: JSON.parse(game.view()) }
      }
      if (view.status === 'Arrived' || view.status === 'Failed') break
      if (view.pending_event)
        apply({
          Respond: {
            event_id: view.pending_event.id,
            choice_id: view.pending_event.choices.find(
              (choice: { available: boolean }) => choice.available,
            ).id,
          },
        })
      else if (view.status.AwaitingRiver)
        apply({ CrossRiver: { method: view.river.ferry_cost_cents !== null ? 'Ferry' : 'Caulk' } })
      else if (view.status.AwaitingFork)
        apply({
          ChooseRoute: {
            route_id: view.routes.find(
              (route: { available: boolean; id: string }) =>
                route.available && route.id !== 'columbia',
            ).id,
          },
        })
      else if (view.inventory.food < 120) apply('Forage')
      else apply(view.status.AtLandmark ? 'Continue' : 'TravelDay')
    }
  }
  throw new Error(`No command-driven ${kind} checkpoint found`)
}

const saves = new Map<string, Awaited<ReturnType<typeof checkpoint>>>()

async function restoreCheckpoint(page: Page, save: string, reducedMotion = false) {
  // Seed storage before React mounts so an older page cannot overwrite the fixture.
  await page.addInitScript(
    ({ save, reducedMotion }) => {
      localStorage.setItem('pioneer-trail:journey:v1', save)
      localStorage.setItem(
        'pioneer-trail:settings:v1',
        JSON.stringify({ sound: false, reducedMotion }),
      )
    },
    { save, reducedMotion },
  )
  await page.goto('/')
}

test('river activities stay open, waiting costs a day, and the crossing resumes', async ({
  page,
}) => {
  const state = saves.get('river') ?? (await checkpoint('river'))
  saves.set('river', state)
  await restoreCheckpoint(page, state.save)
  await expect(page.getByRole('heading', { name: 'The water ahead' })).toBeVisible()
  await page.getByRole('button', { name: 'Wait for better conditions' }).click()
  await expect(page.locator('.journey-heading .eyebrow')).toHaveText(
    `DAY ${state.view.day + 2} OF YOUR JOURNEY`,
  )
  await page.getByRole('button', { name: 'Close panel' }).click()
  await page.getByRole('button', { name: 'Go fishing' }).click()
  await expect(page.getByRole('heading', { name: 'A line in the water' })).toBeVisible()
  await page.getByRole('button', { name: 'Cast your line' }).click()
  await expect(page.locator('.gather-results')).toBeVisible()
  await expect(page.getByRole('heading', { name: 'A line in the water' })).toBeVisible()
  await page.getByRole('button', { name: 'Close panel' }).click()
  await page.getByRole('button', { name: 'Scout the crossing' }).click()
  await page.getByRole('button', { name: 'Take the ferry' }).click()
  await expect(page.getByRole('dialog')).toBeHidden()
  await expect(page.getByRole('button', { name: 'Travel one day' })).toBeVisible()
})

test('a saved wagon breakdown resumes into repair and consumes an original spare', async ({
  page,
}) => {
  const state = saves.get('repair') ?? (await checkpoint('repair'))
  saves.set('repair', state)
  await restoreCheckpoint(page, state.save)
  await expect(page.getByRole('heading', { name: 'Keep the wheels turning' })).toBeVisible()
  await page.getByRole('button', { name: 'Attempt the repair' }).click()
  await expect(page.locator('.journey-heading .eyebrow')).toHaveText(
    `DAY ${state.view.day + 2} OF YOUR JOURNEY`,
  )
  await expect
    .poll(async () =>
      page.evaluate(() => {
        const save = localStorage.getItem('pioneer-trail:journey:v1')!
        // Test-only inspection; never parse and re-save a Rust journey.
        return JSON.parse(save).pending_event
      }),
    )
    .toBeNull()
})

test('the Columbia raft can be steered and completed through the original timed course', async ({
  page,
}) => {
  const state = saves.get('raft') ?? (await checkpoint('raft'))
  saves.set('raft', state)
  await restoreCheckpoint(page, state.save, true)
  await expect(page.getByRole('heading', { name: 'Down the Columbia' })).toBeVisible()
  await page.getByRole('button', { name: '1. Left', exact: true }).click()
  for (let second = 0; second < 30; second++)
    await page.getByRole('button', { name: 'Wait 1 second', exact: true }).click()
  await expect(page.locator('.minigame header')).toContainText('0 seconds')
  await page.getByRole('button', { name: 'Finish crossing', exact: true }).click()
  await expect(page.getByRole('heading', { name: 'You made it west.' })).toBeVisible()
})
